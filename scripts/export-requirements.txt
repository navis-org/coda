#
# This file exists to be a file: `actions/setup-python`'s pip cache is keyed on a
# dependency manifest and fails outright when it can find none. Naming it here also
# keeps the versions the checker resolved against in the repository rather than in a
# workflow step.
neuprint-python
navis

# explicitly because `probe-cave-helpers.py` *runs* the generated helpers rather than only
# resolving their names — the one check here that executes anything.
caveclient
pandas
